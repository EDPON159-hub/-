
-- ЧАСТЬ 1. Создание структуры БД "Магазин запчастей"

-- 1. Создание бд
CREATE DATABASE IF NOT EXISTS AutoPartsStore;
USE AutoPartsStore;

-- 2. Создание таблиц и связей

-- Таблица countries (Страны)
CREATE TABLE countries (
    country_id VARCHAR(10) PRIMARY KEY,
    country_name VARCHAR(100) NOT NULL
);

-- Таблица categories (Категории)
CREATE TABLE categories (
    category_id VARCHAR(10) PRIMARY KEY,
    category_name VARCHAR(100) NOT NULL
);

-- Таблица appliance_types (Тип техники)
CREATE TABLE appliance_types (
    type_id VARCHAR(10) PRIMARY KEY,
    type_name VARCHAR(100) NOT NULL
);

-- Таблица part_names (Названия запчастей)
CREATE TABLE part_names (
    part_name_id VARCHAR(10) PRIMARY KEY,
    part_name VARCHAR(200) NOT NULL,
    category_id VARCHAR(10) NOT NULL,
    FOREIGN KEY (category_id) REFERENCES categories(category_id) ON DELETE CASCADE
);

-- Таблица shops (Магазины) 
CREATE TABLE shops (
    shop_id VARCHAR(20) PRIMARY KEY,
    shop_name VARCHAR(150) NOT NULL,
    address VARCHAR(250) NOT NULL,
    manager VARCHAR(20),
    open_date DATE NOT NULL,
    status VARCHAR(50) NOT NULL DEFAULT 'Работает'
);

-- Таблица employees (Сотрудники)
CREATE TABLE employees (
    employee_id VARCHAR(20) PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    position VARCHAR(100) NOT NULL,
    shop_id VARCHAR(20) NOT NULL,
    phone VARCHAR(20),
    email VARCHAR(100),
    hire_date DATE NOT NULL,
    status VARCHAR(50) NOT NULL DEFAULT 'Работает',
    FOREIGN KEY (shop_id) REFERENCES shops(shop_id) ON DELETE CASCADE
);


ALTER TABLE shops
ADD FOREIGN KEY (manager) REFERENCES employees(employee_id) ON DELETE SET NULL;

-- Таблица customers (Клиенты)
CREATE TABLE customers (
    customer_id VARCHAR(20) PRIMARY KEY,
    customer_name VARCHAR(200) NOT NULL,
    inn VARCHAR(20),
    contact_person VARCHAR(150),
    phone VARCHAR(20),
    email VARCHAR(100),
    address VARCHAR(250),
    status VARCHAR(50) NOT NULL DEFAULT 'Активный'
);

-- Таблица spare_parts (Запчасти)
CREATE TABLE spare_parts (
    part_id VARCHAR(20) PRIMARY KEY,
    part_name_id VARCHAR(10) NOT NULL,
    type_id VARCHAR(10) NOT NULL,
    characteristic VARCHAR(250),
    qty_per_package INT NOT NULL DEFAULT 1,
    country_id VARCHAR(10) NOT NULL,
    base_price DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (part_name_id) REFERENCES part_names(part_name_id) ON DELETE CASCADE,
    FOREIGN KEY (type_id) REFERENCES appliance_types(type_id) ON DELETE CASCADE,
    FOREIGN KEY (country_id) REFERENCES countries(country_id) ON DELETE CASCADE
);

-- Таблица orders (Заказы)
CREATE TABLE orders (
    order_id INT PRIMARY KEY,
    shop_id VARCHAR(20) NOT NULL,
    customer_id VARCHAR(20) NOT NULL,
    employee_id VARCHAR(20) NOT NULL,
    order_date DATE NOT NULL,
    status VARCHAR(50) NOT NULL DEFAULT 'Новый',
    assignment_date DATE,
    planned_ready_date DATE,
    priority VARCHAR(50) DEFAULT 'Стандартный',
    close_date DATE,
    payment_method VARCHAR(50),
    customer_comment TEXT,
    FOREIGN KEY (shop_id) REFERENCES shops(shop_id) ON DELETE CASCADE,
    FOREIGN KEY (customer_id) REFERENCES customers(customer_id) ON DELETE CASCADE,
    FOREIGN KEY (employee_id) REFERENCES employees(employee_id) ON DELETE CASCADE
);

-- Таблица order_items (Состав заказа / позиции заказа)
-- Составной первичный ключ (order_id + part_id)
CREATE TABLE order_items (
    order_id INT NOT NULL,
    part_id VARCHAR(20) NOT NULL,
    qty_sold INT NOT NULL,
    sale_price DECIMAL(10,2) NOT NULL,
    write_off_date DATE,
    warehouse_comment TEXT,
    PRIMARY KEY (order_id, part_id),
    FOREIGN KEY (order_id) REFERENCES orders(order_id) ON DELETE CASCADE,
    FOREIGN KEY (part_id) REFERENCES spare_parts(part_id) ON DELETE CASCADE
);

-- 1)        запросы на выборку (SELECT):where и order by
SELECT 
    sp.part_id AS 'Артикул',
    pn.part_name AS 'Название запчасти',
    c.country_name AS 'Страна производства',
    sp.base_price AS 'Базовая цена'
FROM spare_parts sp, part_names pn, countries c
WHERE sp.part_name_id = pn.part_name_id
  AND sp.country_id = c.country_id
  AND c.country_id = '643'
  AND sp.base_price > 1000
ORDER BY sp.base_price DESC;
--  показать заказы со статусом Выдан или Закрыт
SELECT 
    o.order_id AS 'Номер заказа',
    o.order_date AS 'Дата заказа',
    o.status AS 'Статус',
    c.customer_name AS 'Клиент',
    e.full_name AS 'Сотрудник'
FROM orders o, customers c, employees e
WHERE o.customer_id = c.customer_id
  AND o.employee_id = e.employee_id
  AND o.status IN ('Выдан', 'Закрыт')
ORDER BY o.order_date DESC;
-- 2   явное соединение таблиц  (INNER JOIN)
SELECT 
    sp.part_id AS 'Артикул',
    pn.part_name AS 'Название',
    cat.category_name AS 'Категория',
    at.type_name AS 'Тип техники',
    c.country_name AS 'Страна',
    sp.characteristic AS 'Характеристики',
    sp.base_price AS 'Цена'
FROM spare_parts sp
INNER JOIN part_names pn ON sp.part_name_id = pn.part_name_id
INNER JOIN categories cat ON pn.category_id = cat.category_id
INNER JOIN countries c ON sp.country_id = c.country_id
ORDER BY cat.category_name, sp.base_price DESC;
-- (LEFT JOIN)
SELECT 
    o.order_id AS '№ Заказа',
    o.order_date AS 'Дата',
    s.shop_name AS 'Магазин',
    c.customer_name AS 'Клиент',
    e.full_name AS 'Продавец',
    o.status AS 'Статус',
    o.priority AS 'Приоритет'
FROM orders o
INNER JOIN shops s ON o.shop_id = s.shop_id
INNER JOIN customers c ON o.customer_id = c.customer_id
INNER JOIN employees e ON o.employee_id = e.employee_id
ORDER BY o.order_date DESC;
-- 3  агрегирующие операции (COUNT, SUM)
SELECT 
    COUNT(DISTINCT o.order_id) AS 'Всего заказов',
    COUNT(oi.order_id) AS 'Всего позиций в заказах',
    SUM(oi.qty_sold) AS 'Общее количество проданных товаров',
    SUM(oi.qty_sold * oi.sale_price) AS 'Общая сумма продаж'
FROM orders o
LEFT JOIN order_items oi ON o.order_id = oi.order_id;
-- (AVG, MAX, MIN)
SELECT 
    COUNT(*) AS 'Всего наименований запчастей',
    AVG(base_price) AS 'Средняя цена',
    MAX(base_price) AS 'Максимальная цена',
    MIN(base_price) AS 'Минимальная цена'
FROM spare_parts;

-- 4 группировку GROUP BY
SELECT 
    status AS 'Статус заказа',
    COUNT(*) AS 'Количество заказов'
FROM orders
GROUP BY status
ORDER BY COUNT(*) DESC;
-- фильтрацию HAVING
SELECT 
    s.shop_id AS 'ID магазина',
    s.shop_name AS 'Название магазина',
    COUNT(DISTINCT o.order_id) AS 'Количество заказов',
    SUM(oi.qty_sold * oi.sale_price) AS 'Общая сумма продаж'
FROM shops s
INNER JOIN orders o ON s.shop_id = o.shop_id
INNER JOIN order_items oi ON o.order_id = oi.order_id
GROUP BY s.shop_id, s.shop_name
HAVING SUM(oi.qty_sold * oi.sale_price) > 3000
ORDER BY SUM(oi.qty_sold * oi.sale_price) DESC; 


-- 1. Вложенный запрос с оператором IN

SELECT
    sp.part_id AS 'Артикул',
    pn.part_name AS 'Название запчасти',
    sp.base_price AS 'Цена'
FROM spare_parts sp
INNER JOIN part_names pn
    ON sp.part_name_id = pn.part_name_id
WHERE sp.part_id IN (
    SELECT oi.part_id
    FROM order_items oi
    INNER JOIN orders o
        ON oi.order_id = o.order_id
    WHERE o.status IN ('Закрыт', 'Выдан')
);

-- 2. Вложенный запрос с оператором ALL

SELECT
    sp.part_id AS 'Артикул',
    pn.part_name AS 'Название запчасти',
    sp.base_price AS 'Цена'
FROM spare_parts sp
INNER JOIN part_names pn
    ON sp.part_name_id = pn.part_name_id
WHERE sp.base_price > ALL (
    SELECT sp2.base_price
    FROM spare_parts sp2
    INNER JOIN part_names pn2
        ON sp2.part_name_id = pn2.part_name_id
    INNER JOIN categories cat
        ON pn2.category_id = cat.category_id
    WHERE cat.category_name = 'Насосы'
);


--Функции управления потоком IF и CASE
SELECT
    sp.part_id AS 'Артикул',
    pn.part_name AS 'Название',
    sp.base_price AS 'Цена',

    IF(
        sp.base_price >= 2000,
        'Дорогая запчасть',
        'Бюджетная запчасть'
    ) AS 'Ценовая категория'

FROM spare_parts sp
INNER JOIN part_names pn
    ON sp.part_name_id = pn.part_name_id;

--Определение состояния заказа через CASE.
SELECT
    order_id AS 'Номер заказа',
    status AS 'Текущий статус',

    CASE
        WHEN status = 'Новый'
            THEN 'Требуется обработка'

        WHEN status = 'В обработке'
            THEN 'Заказ собирается'

        WHEN status = 'Выдан'
            THEN 'Заказ выдан клиенту'

        WHEN status = 'Закрыт'
            THEN 'Заказ завершён'

        ELSE 'Неизвестный статус'
    END AS 'Описание состояния'

FROM orders;

--Временное представление WITH
WITH shop_sales AS (

    SELECT
        s.shop_id,
        s.shop_name,
        COUNT(DISTINCT o.order_id) AS order_count,
        SUM(oi.qty_sold * oi.sale_price) AS total_sales

    FROM shops s

    LEFT JOIN orders o
        ON s.shop_id = o.shop_id

    LEFT JOIN order_items oi
        ON o.order_id = oi.order_id

    GROUP BY s.shop_id, s.shop_name

)

SELECT
    shop_name AS 'Магазин',
    order_count AS 'Количество заказов',
    COALESCE(total_sales, 0) AS 'Общая выручка'

FROM shop_sales
ORDER BY total_sales DESC;
